-- Detailed table structure with primary keys, foreign keys, and constraints
WITH table_columns AS (
    SELECT 
        table_name,
        json_agg(
            json_build_object(
                'column_name', column_name,
                'data_type', data_type,
                'character_maximum_length', character_maximum_length,
                'numeric_precision', numeric_precision,
                'is_nullable', is_nullable,
                'column_default', column_default
            ) ORDER BY ordinal_position
        ) as columns
    FROM information_schema.columns
    WHERE table_schema = 'public' 
        AND table_name IN ('users', 'verification_codes', 'admin_users', 'admin_sessions')
    GROUP BY table_name
),
table_constraints AS (
    SELECT 
        tc.table_name,
        json_agg(
            json_build_object(
                'constraint_name', tc.constraint_name,
                'constraint_type', tc.constraint_type,
                'column_name', kcu.column_name
            )
        ) as constraints
    FROM information_schema.table_constraints tc
    LEFT JOIN information_schema.key_column_usage kcu
        ON tc.constraint_name = kcu.constraint_name
        AND tc.table_schema = kcu.table_schema
    WHERE tc.table_schema = 'public'
        AND tc.table_name IN ('users', 'verification_codes', 'admin_users', 'admin_sessions')
    GROUP BY tc.table_name
),
table_indexes AS (
    SELECT 
        tablename as table_name,
        json_agg(
            json_build_object(
                'index_name', indexname,
                'index_definition', indexdef
            )
        ) as indexes
    FROM pg_indexes
    WHERE schemaname = 'public'
        AND tablename IN ('users', 'verification_codes', 'admin_users', 'admin_sessions')
    GROUP BY tablename
)
SELECT 
    tc.table_name,
    tc.columns,
    COALESCE(tcon.constraints, '[]'::json) as constraints,
    COALESCE(ti.indexes, '[]'::json) as indexes
FROM table_columns tc
LEFT JOIN table_constraints tcon ON tc.table_name = tcon.table_name
LEFT JOIN table_indexes ti ON tc.table_name = ti.table_name
ORDER BY tc.table_name;
